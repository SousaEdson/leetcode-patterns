class Solution {

    private Map<Integer, List<Integer>> buildNodesPerIndex(TreeNode root){

        Map<Integer, List<Integer>> nodesPerIndex = new HashMap<>();
        Queue<TreeNode> queue = new LinkedList<>();
        Map<TreeNode, Integer> nodePosition = new HashMap<>();

        queue.offer(root);
        nodePosition.put(root, 0);
        
        while(!queue.isEmpty()){            
            int length = queue.size();
            Map<Integer, List<Integer>> nodes = new HashMap<>(); 
            for(int i=0; i<length; i++){
                TreeNode currentNode = queue.poll();
                int currentPosition = nodePosition.get(currentNode);                
                if(nodes.get(currentPosition) == null) nodes.put(currentPosition, new ArrayList<>());
                nodes.get(currentPosition).add(currentNode.val);

                if(currentNode.left != null){
                    queue.offer(currentNode.left);
                    nodePosition.put(currentNode.left, currentPosition - 1);                    
                }
                if(currentNode.right != null){
                    queue.offer(currentNode.right);
                    nodePosition.put(currentNode.right, currentPosition + 1);                    
                }
            }

            for(Integer position: nodes.keySet()){
                List<Integer> nodeList = nodes.get(position);
                nodeList.sort(null);
                if(nodesPerIndex.get(position) == null) nodesPerIndex.put(position, new ArrayList<>());
                nodesPerIndex.get(position).addAll(nodeList);
            }
        }
        return nodesPerIndex;
    }

    public List<List<Integer>> verticalTraversal(TreeNode root) {
        List<List<Integer>> nodesInVerticalOrder = new ArrayList<>();
        Map<Integer, List<Integer>> nodesPerIndex = buildNodesPerIndex(root);
        
        List<Integer> indexes = new ArrayList<>(nodesPerIndex.keySet());
        indexes.sort(null);

        for(Integer index: indexes){
            nodesInVerticalOrder.add(nodesPerIndex.get(index));
        }

        return nodesInVerticalOrder;
    }
}
